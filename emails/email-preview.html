Intégration Resend

Créez un compte sur resend.com (gratuit jusqu'à 3 000 emails/mois), récupérez votre API key, et ajoutez-la comme secret Supabase. 

// Dans chaque Edge Function Supabase
import { Resend } from 'npm:resend'

const resend = new Resend(Deno.env.get('RESEND_API_KEY'))

// Charger le template HTML
const html = await Deno.readTextFile('./email-01-bienvenue.html')
  .replace(/\{\{prenom\}\}/g, prenom)
  .replace(/\{\{unsubscribeToken\}\}/g, token)

await resend.emails.send({
  from: 'Blooom <hello@blooom.app>',
  to: [email],
  subject: 'Bienvenue dans Blooom 🌸',
  html,
})

// Ajouter le secret dans Supabase Dashboard :
// Settings → Edge Functions → Secrets → RESEND_API_KEY
